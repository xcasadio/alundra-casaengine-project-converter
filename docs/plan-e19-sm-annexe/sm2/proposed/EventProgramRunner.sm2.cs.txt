    /// <summary>
    /// The test of 0x6B/0x6C/0x6D (E19.sm2, SM2-R2), the handlers' own instructions (<c>0x8003F908-0x8003F948</c>): <c>subu</c> (wraps), <c>bgez</c>/<c>negu</c> (the
    /// negation wraps: <c>int.MinValue</c> stays <c>int.MinValue</c>, hence NO <c>Math.Abs</c>, which throws on it), the threshold <c>(v1 | v2 &lt;&lt; 8) &lt;&lt; 16</c> as a 32-bit
    /// signed integer (negative from <c>v2</c> = 0x80, so such a threshold is below every distance but <c>int.MinValue</c>), <c>slt</c> (signed) and <c>xori 1</c>:
    /// the result is "distance not below threshold". <c>unchecked</c> is explicit although no project turns overflow checking on (FAIT: no <c>CheckForOverflowUnderflow</c>
    /// and no <c>checked</c> keyword in the repository).
    /// </summary>
    private static bool MovedAtLeastTheThreshold(int memo, int position, int[] v)
    {
        unchecked
        {
            var distance = memo - position;
            if (distance < 0)
            {
                distance = -distance;
            }

            var threshold = (v[1] | (v[2] << 8)) << 16;
            return !(distance < threshold);
        }
    }
