<Project Sdk="Microsoft.NET.Sdk">

  <!--
    O-E19-63 frame-time harness. Out of the repository on purpose: it references COPIES (lib-debug / lib-release,
    taken from Alundra.Tests/bin/<cfg>/net9.0-windows) of the repository's own built assemblies by HintPath. No
    ProjectReference to any repository project, so nothing in the repository is built or written.
    Mirrors CasaEngine.Launcher.csproj: RollForward Major, TieredCompilation false, UseWindowsForms.
  -->
  <PropertyGroup>
    <OutputType>WinExe</OutputType>
    <TargetFramework>net9.0-windows</TargetFramework>
    <RollForward>Major</RollForward>
    <TieredCompilation>false</TieredCompilation>
    <UseWindowsForms>true</UseWindowsForms>
    <ImplicitUsings>enable</ImplicitUsings>
    <Nullable>disable</Nullable>
    <AssemblyName>O63Harness</AssemblyName>
    <RootNamespace>O63Harness</RootNamespace>
    <!-- No deps.json: the host then probes every assembly of the application directory (the copied set). -->
    <GenerateDependencyFile>false</GenerateDependencyFile>
    <LibDir Condition="'$(Configuration)' == 'Release'">$(MSBuildProjectDirectory)\..\lib-release</LibDir>
    <LibDir Condition="'$(Configuration)' != 'Release'">$(MSBuildProjectDirectory)\..\lib-debug</LibDir>
  </PropertyGroup>

  <ItemGroup>
    <Reference Include="CasaEngine"><HintPath>$(LibDir)\CasaEngine.dll</HintPath><Private>false</Private></Reference>
    <Reference Include="MonoGame.Framework"><HintPath>$(LibDir)\MonoGame.Framework.dll</HintPath><Private>false</Private></Reference>
    <Reference Include="MGUI.MonoGame.Integration"><HintPath>$(LibDir)\MGUI.MonoGame.Integration.dll</HintPath><Private>false</Private></Reference>
    <Reference Include="MGUI.Core"><HintPath>$(LibDir)\MGUI.Core.dll</HintPath><Private>false</Private></Reference>
    <Reference Include="MGUI.Shared"><HintPath>$(LibDir)\MGUI.Shared.dll</HintPath><Private>false</Private></Reference>
    <Reference Include="MGUI.Rendering.Abstractions"><HintPath>$(LibDir)\MGUI.Rendering.Abstractions.dll</HintPath><Private>false</Private></Reference>
  </ItemGroup>

  <!-- Copy the whole engine set next to the exe, except the gameplay assembly and the test assemblies; the gameplay
       assembly goes to gameplay\ and is loaded with Assembly.LoadFile like the launcher does (AssemblyManager). -->
  <Target Name="CopyEngineSet" AfterTargets="Build">
    <ItemGroup>
      <EngineFile Include="$(LibDir)\**\*.*" Exclude="$(LibDir)\Alundra.dll;$(LibDir)\Alundra.pdb;$(LibDir)\Alundra.Tests.*;$(LibDir)\testhost*;$(LibDir)\xunit*;$(LibDir)\Microsoft.TestPlatform*;$(LibDir)\Microsoft.VisualStudio.*;$(LibDir)\*.runtimeconfig.json;$(LibDir)\*.deps.json" />
      <GameplayFile Include="$(LibDir)\Alundra.dll;$(LibDir)\Alundra.pdb" />
      <!-- Without a deps.json the host does not map runtimes\win-x64\native: put openal.dll and SDL2.dll at the root. -->
      <NativeFile Include="$(LibDir)\runtimes\win-x64\native\*.dll" />
    </ItemGroup>
    <Copy SourceFiles="@(NativeFile)" DestinationFolder="$(OutDir)" SkipUnchangedFiles="true" />
    <Copy SourceFiles="@(EngineFile)" DestinationFiles="@(EngineFile->'$(OutDir)%(RecursiveDir)%(Filename)%(Extension)')" SkipUnchangedFiles="true" />
    <Copy SourceFiles="@(GameplayFile)" DestinationFolder="$(OutDir)gameplay" SkipUnchangedFiles="true" />
  </Target>

</Project>
