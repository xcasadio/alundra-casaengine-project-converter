/// <summary>SCRATCH PROBE (op22-disc, never committed): runs the real DLL at the 0x22/0x23 sites and writes, per site, the frames of the wait and the Z states.</summary>
[Collection(AlundraMusicPlayerSingletonCollection.Name)]
public sealed class Op22ProbeTests
{
    private const string OutDir = @"<scratchpad>\op22-disc\probe-out";

    internal sealed record Site(string Id, int Map, int[] Flags, int HeroX, int HeroY, int HeroZ, int Rec, int Slot, int Pc, int Opcode, int Frames)
    {
        public bool RideHero { get; init; }
        public bool Dialogue { get; init; }
    }

    private static IEnumerable<Site> AllSites()
    {
        // 127 balls: hero in the gate of the program (3B operands), no overlap with the balls
        yield return new Site("s127_rec17", 127, Array.Empty<int>(), 5, 40, 13, 17, 2, 640, 0x22, 2500);
        yield return new Site("s127_rec18", 127, Array.Empty<int>(), 10, 40, 13, 18, 2, 676, 0x22, 2500);
        yield return new Site("s127_rec6", 127, Array.Empty<int>(), 10, 48, 5, 6, 2, 280, 0x22, 2500);
        yield return new Site("s127_rec7", 127, Array.Empty<int>(), 15, 48, 5, 7, 2, 316, 0x22, 2500);
        yield return new Site("s127_rec8", 127, Array.Empty<int>(), 3, 44, 9, 8, 2, 352, 0x22, 2500);
        yield return new Site("s127_rec9", 127, Array.Empty<int>(), 5, 44, 9, 9, 2, 388, 0x22, 2500);
        yield return new Site("s127_rec12", 127, Array.Empty<int>(), 4, 50, 1, 12, 2, 424, 0x22, 2500);
        yield return new Site("s127_rec11", 127, Array.Empty<int>(), 7, 50, 1, 11, 2, 460, 0x22, 2500);
        yield return new Site("s127_rec13", 127, Array.Empty<int>(), 4, 44, 9, 13, 2, 496, 0x22, 2500);
        yield return new Site("s127_rec14", 127, Array.Empty<int>(), 7, 44, 9, 14, 2, 532, 0x22, 2500);
        yield return new Site("s127_rec15", 127, Array.Empty<int>(), 10, 44, 9, 15, 2, 568, 0x22, 2500);
        yield return new Site("s127_rec16", 127, Array.Empty<int>(), 13, 44, 9, 16, 2, 604, 0x22, 2500);

        // 89: balls (gravity fall of 64 px then rise)
        foreach (var r in new[] { 20, 21, 22, 23, 24, 25 })
        {
            yield return new Site("s89_rec" + r, 89, Array.Empty<int>(), 20, 35, 3, r, 2, 663, 0x22, 3000);
        }

        // 363: iron grid
        yield return new Site("s363_rec0", 363, Array.Empty<int>(), 10, 10, 3, 0, 2, 454, 0x22, 3000);
        // 36: the 12 hanging ceilings (0x23 @1252)
        foreach (var r in Enumerable.Range(42, 12))
        {
            yield return new Site("s36c14_rec" + r, 36, Array.Empty<int>(), 5, 10, 1, r, 2, 1252, 0x23, 5000);
        }

        // 36 C[4]: the floating platform, the hero rides it (placed on it once landed)
        yield return new Site("s36c4_rec3", 36, Array.Empty<int>(), 30, 19, 1, 3, 2, 491, 0x22, 3000) { RideHero = true };
        // 115 B[2]: the block of the descent (full real program of the map event)
        yield return new Site("s115_rec19", 115, Array.Empty<int>(), 10, 55, 0, 19, 1, 367, 0x22, 6000) { Dialogue = true };
    }

    public static IEnumerable<object[]> SiteIds() => AllSites().Select(s => new object[] { s.Id });

    [Theory]
    [MemberData(nameof(SiteIds))]
    public void Probe(string id)
    {
        var site = AllSites().Single(s => s.Id == id);
        Directory.CreateDirectory(OutDir);
        var root = SaveGameDirectorTestSupport.FindProjectRoot();
        var mapsDir = Path.Combine(root, "Maps");
        var worldDir = Directory.GetDirectories(mapsDir).SelectMany(Directory.GetDirectories)
            .First(d => Path.GetFileName(d).EndsWith("-" + site.Map, StringComparison.Ordinal));
        var world = Path.GetFileName(worldDir);
        var zone = Path.GetFileName(Path.GetDirectoryName(worldDir)!);

        using var arc = new ArcRun(new ArcSpec("Op22-" + site.Id, zone, world, site.Flags, site.HeroX, site.HeroY, site.HeroZ, site.Frames, RealController: true, Prefabs: true,
            Arrival: new ArcArrival((site.HeroX * 24 + 12) << 16, (site.HeroY * 16 + 8) << 16, 1 << 20, AlundraGameState.ResetAnimationId, 0)));
        // the map load clears the temporary flags (bit 0x8000 banks): the flags the programs wait for are set after it
        if (site.Id == "s363_rec0") { AlundraGameState.Instance.AddFlag(32769, 1u << 1); }
        if (site.RideHero) { AlundraGameState.Instance.AddFlag(32774, 1u << 6); }
        var framesAtPc = new List<int>();
        var endFrame = -1;
        var states = new List<(int Frame, int PosZ, int ForceZ, int Final, int Ground, int Contact)>();
        arc.OnInstruction = t =>
        {
            if (t.Slot == site.Slot && t.Pc == site.Pc && t.Opcode == site.Opcode)
            {
                framesAtPc.Add(t.Frame);
            }
            else if (t.Slot == site.Slot && t.Pc == site.Pc + 1 && framesAtPc.Count > 0 && endFrame < 0)
            {
                endFrame = t.Frame;
            }
        };
        var ridden = false;
        arc.OnFrame = () =>
        {
            var e = arc.EntityByRecord(site.Rec);
            if (site.RideHero && arc.Frame == 3)
            {
                AlundraGameState.Instance.SetFlag(32774, ~(1u << 6));
            }

            if (site.RideHero && !ridden && e != null && e.CollidedWithEntityZ != 0 && e.PosZ <= (16 << 16) + 70000)
            {
                ridden = true;
                arc.PlaceHero((e.PosX >> 16), (e.PosY >> 16), (e.PosZ >> 16) + 8);
            }

            if (e != null && framesAtPc.Count > 0)
            {
                states.Add((arc.Frame - 1, e.PosZ, e.ForceZ, e.FinalForceZ, e.IsOnGround, e.CollidedWithEntityZ));
            }
        };

        var sb = new StringBuilder();
        try
        {
            if (site.Dialogue)
            {
                arc.RunUntilPressingTheButtonOnEveryDialogueFrame(() => endFrame >= 0, "the 0x22 wait ends");
            }
            else
            {
                arc.RunUntil(() => endFrame >= 0, "the 0x22 wait ends");
            }
        }
        catch (Exception ex)
        {
            sb.AppendLine("STUCK: " + ex.Message);
        }

        sb.AppendLine("spawned: " + string.Join(", ", arc.Entities.Select(e => $"#{e.EntityRefId}@{e.PosZ >> 16}")));
        if (arc.Log != null) { foreach (var l in arc.Log.Infos.Concat(arc.Log.Warnings).Concat(arc.Log.Errors).Take(40)) { sb.AppendLine("LOG " + l); } }
        var first = framesAtPc.Count > 0 ? framesAtPc[0] : -1;
        var last = framesAtPc.Count > 0 ? framesAtPc[^1] : -1;
        sb.AppendLine($"site {site.Id}: map {site.Map} rec {site.Rec} slot {site.Slot} pc {site.Pc} op 0x{site.Opcode:X2}");
        sb.AppendLine($"records at the wait pc: {framesAtPc.Count}; first frame {first}; last frame {last}; end (pc+1) frame {endFrame}; calls after the first (end - first) = {endFrame - first}; (last - first) = {last - first}");
        var ent = arc.EntityByRecord(site.Rec);
        sb.AppendLine($"entity: {(ent == null ? "none" : $"PosZ {ent.PosZ} Height {ent.RecordHeight} target-literal {(ent.RecordHeight ?? 0) << 19}")}");
        foreach (var s in states.Take(3).Concat(states.TakeLast(6)))
        {
            sb.AppendLine($"  frame {s.Frame}: PosZ {s.PosZ} ForceZ {s.ForceZ} Final {s.Final} OnGround {s.Ground} Contact {s.Contact}");
        }

        var rule = Environment.GetEnvironmentVariable("OP22_RULE") ?? "lit";
        var tag = Environment.GetEnvironmentVariable("OP22_TAG") ?? "";
        File.WriteAllText(Path.Combine(OutDir, site.Id + "." + rule + tag + ".txt"), sb.ToString());
        var lastState = states.Count > 0 ? states[^1] : default;
        var calls = endFrame >= 0 ? (endFrame - first).ToString() : "STUCK";
        File.AppendAllText(Path.Combine(OutDir, "summary." + rule + tag + ".tsv"),
            string.Join("\t", site.Id, calls, framesAtPc.Count, lastState.PosZ, lastState.ForceZ) + "\n");
        Assert.True(endFrame >= 0, sb.ToString());
    }

    public static IEnumerable<object[]> CycleIds() => new[] { new object[] { "rec6" }, new object[] { "rec17" }, new object[] { "rec8" } };

    [Theory]
    [MemberData(nameof(CycleIds))]
    public void Cycles(string which)
    {
        var (site, entry, pcs) = which switch
        {
            "rec6" => (AllSites().Single(x => x.Id == "s127_rec6"), 256, new[] { 272, 280, 285 }),
            "rec17" => (AllSites().Single(x => x.Id == "s127_rec17"), 616, new[] { 632, 640, 645 }),
            _ => (AllSites().Single(x => x.Id == "s127_rec8"), 328, new[] { 344, 352, 357 }),
        };
        var root = SaveGameDirectorTestSupport.FindProjectRoot();
        var worldDir = Directory.GetDirectories(Path.Combine(root, "Maps")).SelectMany(Directory.GetDirectories)
            .First(d => Path.GetFileName(d).EndsWith("-" + site.Map, StringComparison.Ordinal));
        using var arc = new ArcRun(new ArcSpec("Op22c-" + which, Path.GetFileName(Path.GetDirectoryName(worldDir)!), Path.GetFileName(worldDir), site.Flags,
            site.HeroX, site.HeroY, site.HeroZ, 1400, RealController: true, Prefabs: true,
            Arrival: new ArcArrival((site.HeroX * 24 + 12) << 16, (site.HeroY * 16 + 8) << 16, 1 << 20, AlundraGameState.ResetAnimationId, 0)));
        var marks = new List<string>();
        int prevPc = -1;
        arc.OnInstruction = t =>
        {
            if (t.Slot == 2 && t.ProgramStart == entry && pcs.Contains(t.Pc) && t.Pc != prevPc)
            {
                marks.Add($"{t.Frame}:{t.Pc}");
            }

            if (t.Slot == 2 && t.ProgramStart == entry)
            {
                prevPc = t.Pc;
            }
        };
        for (var i = 0; i < 1000; i++)
        {
            arc.OneFrame();
        }

        var rule = Environment.GetEnvironmentVariable("OP22_RULE") ?? "lit";
        var tag = Environment.GetEnvironmentVariable("OP22_TAG") ?? "";
        File.WriteAllText(Path.Combine(OutDir, "cycles." + which + "." + rule + tag + ".txt"), string.Join(" ", marks) + Environment.NewLine);
    }
}
