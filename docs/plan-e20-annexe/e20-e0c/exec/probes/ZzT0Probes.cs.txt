/// <summary>E20.e0c T0 probes (scratch, never committed: copied to docs/plan-e20-annexe/e20-e0c/exec/probes/). Lines are prefixed "T0:".</summary>
[Collection(MusicPlayerSingletonCollection.Name)]

{
    private const int Arrow = 3;
    private const int CoinIndex = 99; // item 0x45 + 0x1E
    private readonly ITestOutputHelper _output;

    public ZzT0Probes(ITestOutputHelper output) => _output = output;

    private void Say(string line) => _output.WriteLine("T0: " + line);

    private static ArcRun Arc14(string name) => new(new ArcSpec("E20e0c-" + name, "Overworld", "Overworld 3,1-14", Array.Empty<int>(), 8, 6, 0, 2000, RealController: true, Prefabs: true));

    private static IEntityWorldContext Context(ArcRun arc) => arc.Proxy;

    private sealed class Spy : IEventProgramRunner
    {
        private readonly IEventProgramRunner _inner;
        private readonly Action<EntityScriptProxy> _before;

        public Spy(IEventProgramRunner inner, Action<EntityScriptProxy> before)
        {
            _inner = inner;
            _before = before;
        }

        public void RunScript(EntityScriptProxy entity, int programSlot) => _inner.RunScript(entity, programSlot);

        public void RunSpriteEvent(EntityScriptProxy entity)
        {
            _before(entity);
            _inner.RunSpriteEvent(entity);
        }
    }

    private sealed class Rig
    {
        public EntityScriptProxy? Child;
        public EntityScriptProxy? Carrier;
        public int Tick;
        public int CarrierTick = -1;
        public bool Spawned;
        public readonly List<string> Log = new();
        public int FirstChildProgramTick = -1;
        public int FirstChildCTick = -1;
        public int FirstChildIsOnGround = -99;
    }

    private (ArcRun Arc, Rig Rig) Mount(string name, bool childLower)
    {
        var arc = Arc14(name);
        var rig = new Rig();
        arc.Proxy.EventProgramRunner = new Spy(arc.Proxy.EventProgramRunner, entity =>
        {
            if (entity.SpriteType == Arrow && !rig.Spawned && ReferenceEquals(entity, rig.Carrier))
            {
                rig.Spawned = true;
                rig.CarrierTick = rig.Tick;
                rig.Child = Context(arc).SpawnEntityBySpriteIndex(entity, false, CoinIndex, entity.PosX, entity.PosY, entity.PosZ, 0);
                if (rig.Child != null)
                {
                    rig.Child.ForceZ = 0xA0000;
                    rig.Child.Flags &= ~0x80u;
                    rig.Log.Add($"tick {rig.Tick}: carrier slot {entity.Slot} spawned child slot {rig.Child.Slot} status {rig.Child.Status} onGround {rig.Child.IsOnGround} posZ {rig.Child.PosZ} forceZ {rig.Child.ForceZ}");
                }
            }
            else if (rig.Child != null && ReferenceEquals(entity, rig.Child))
            {
                if (rig.FirstChildProgramTick < 0)
                {
                    rig.FirstChildProgramTick = rig.Tick;
                    rig.FirstChildIsOnGround = entity.IsOnGround;
                }

                rig.Log.Add($"tick {rig.Tick}: child program trigger {entity.EventTrigger} status {entity.Status} onGround {entity.IsOnGround} posZ {entity.PosZ} forceZ {entity.ForceZ} frameCounter {entity.FrameCounter}");
                if (entity.EventTrigger == ScriptHelper.ProgramCTick && rig.FirstChildCTick < 0)
                {
                    rig.FirstChildCTick = rig.Tick;
                }
            }
        });
        arc.Proxy.TickTraceForTests = pass =>
        {
            if (pass == "pad")
            {
                rig.Tick++;
            }

            if (rig.Child != null && pass == "physics")
            {
                rig.Log.Add($"tick {rig.Tick} [physics start]: child status {rig.Child.Status} onGround {rig.Child.IsOnGround} posZ {rig.Child.PosZ} forceZ {rig.Child.ForceZ} moddedZ {rig.Child.ModdedPosZ}");
            }
        };

        if (childLower)
        {
            var a = Context(arc).SpawnEntityBySpriteIndex(arc.Hero, false, Arrow, 80 << 16, 80 << 16, 0, 0x10)!;
            var b = Context(arc).SpawnEntityBySpriteIndex(arc.Hero, false, Arrow, 90 << 16, 80 << 16, 0, 0x10)!;
            rig.Carrier = Context(arc).SpawnEntityBySpriteIndex(arc.Hero, false, Arrow, 100 << 16, 100 << 16, 0, 0x10)!;
            Say($"slots a={a.Slot} b={b.Slot} carrier={rig.Carrier.Slot}");
            arc.OneFrame(); // integrate: the carrier's A would already spawn here, so hold it: spawned flag set below.
        }
        else
        {
            rig.Carrier = Context(arc).SpawnEntityBySpriteIndex(arc.Hero, false, Arrow, 100 << 16, 100 << 16, 0, 0x10)!;
        }

        return (arc, rig);
    }

    [Theory]
    [InlineData(false)]
    [InlineData(true)]
    public void PL1_PL2_TheDroppedObjectsFirstTicks(bool childLower)
    {
        using var arc = Arc14("pl1");
        var rig = new Rig();
        EntityScriptProxy? fillerA = null;
        arc.Proxy.EventProgramRunner = new Spy(arc.Proxy.EventProgramRunner, entity =>
        {
            if (rig.Carrier != null && ReferenceEquals(entity, rig.Carrier) && entity.EventTrigger == ScriptHelper.ProgramCTick && !rig.Spawned)
            {
                if (childLower && fillerA != null && fillerA.Status != EntityStatus.Destroyed && fillerA.Status != EntityStatus.FlagToDestroy)
                {
                    return; // wait for the gap
                }

                rig.Spawned = true;
                rig.CarrierTick = rig.Tick;
                rig.Child = Context(arc).SpawnEntityBySpriteIndex(entity, false, CoinIndex, entity.PosX, entity.PosY, entity.PosZ, 0);
                if (rig.Child != null)
                {
                    rig.Child.ForceZ = 0xA0000;
                    rig.Child.Flags &= ~0x80u;
                    rig.Log.Add($"tick {rig.Tick}: carrier slot {entity.Slot} spawned child slot {rig.Child.Slot} status {rig.Child.Status} onGround {rig.Child.IsOnGround} posZ {rig.Child.PosZ} forceZ {rig.Child.ForceZ} carrierPosZ {entity.PosZ}");
                }
            }
            else if (rig.Child != null && ReferenceEquals(entity, rig.Child))
            {
                if (rig.FirstChildProgramTick < 0)
                {
                    rig.FirstChildProgramTick = rig.Tick;
                    rig.FirstChildIsOnGround = entity.IsOnGround;
                }

                rig.Log.Add($"tick {rig.Tick}: child program trigger {entity.EventTrigger} status {entity.Status} onGround {entity.IsOnGround} posZ {entity.PosZ} forceZ {entity.ForceZ} frameCounter {entity.FrameCounter}");
                if (entity.EventTrigger == ScriptHelper.ProgramCTick && rig.FirstChildCTick < 0)
                {
                    rig.FirstChildCTick = rig.Tick;
                }
            }
        });
        arc.Proxy.TickTraceForTests = pass =>
        {
            if (pass == "pad")
            {
                rig.Tick++;
            }

            if (rig.Child != null && pass == "physics")
            {
                rig.Log.Add($"tick {rig.Tick} [physics start]: child status {rig.Child.Status} onGround {rig.Child.IsOnGround} posZ {rig.Child.PosZ} forceZ {rig.Child.ForceZ} moddedZ {rig.Child.ModdedPosZ}");
            }
        };

        if (childLower)
        {
            fillerA = Context(arc).SpawnEntityBySpriteIndex(arc.Hero, false, Arrow, 80 << 16, 80 << 16, 0, 0x10)!;
            var filler = Context(arc).SpawnEntityBySpriteIndex(arc.Hero, false, Arrow, 90 << 16, 80 << 16, 0, 0x10)!;
            rig.Carrier = Context(arc).SpawnEntityBySpriteIndex(arc.Hero, false, Arrow, 100 << 16, 100 << 16, 0, 0x10)!;
            Say($"slots fillerA={fillerA.Slot} filler={filler.Slot} carrier={rig.Carrier.Slot}");
            arc.OneFrame();
            Context(arc).DestroyEntity(fillerA);
            arc.Proxy.RecycleDestroyedEntities();
            Say($"fillerA destroyed+recycled; status {fillerA.Status}");
        }
        else
        {
            rig.Carrier = Context(arc).SpawnEntityBySpriteIndex(arc.Hero, false, Arrow, 100 << 16, 100 << 16, 0, 0x10)!;
            Say($"carrier slot {rig.Carrier.Slot}");
        }

        for (var i = 0; i < 6; i++)
        {
            arc.OneFrame();
        }

        Say($"childLower={childLower}: carrierTick={rig.CarrierTick} firstChildProgramTick={rig.FirstChildProgramTick} firstChildCTick={rig.FirstChildCTick} firstChildIsOnGround={rig.FirstChildIsOnGround}");
        foreach (var line in rig.Log)
        {
            Say("  " + line);
        }

        Assert.NotNull(rig.Child);
    }

    private static ArcRun ArcDebug(string name) => new(new ArcSpec("E20e0c-" + name, "Debug zone", "Debug zone-11", Array.Empty<int>(), 54, 30, 0, 2000, RealController: true, Prefabs: true));

    [Fact]
    public void PJ_TheHeroOnAPosedObject()
    {
        foreach (var clear in new[] { false, true })
        {
            using var arc = ArcDebug("pj" + clear);
            arc.OneFrame();
            foreach (var record in new[] { 20, 21, 22, 23, 6 })
            {
                var e = arc.EntityByRecord(record);
                Assert.NotNull(e);
                Say($"clear={clear} record {record}: type {e!.SpriteType} slot {e.Slot} status {e.Status} pos ({e.PosX >> 16},{e.PosY >> 16},{e.PosZ}) mod ({e.ModX >> 16},{e.ModY >> 16},{e.ModZ}) size ({e.Width >> 16},{e.Height >> 16},{e.Depth}) flags {e.Flags:X8} collidable {(e.Flags & 0x80) != 0} onGround {e.IsOnGround}");
            }

            var target = arc.EntityByRecord(22)!;
            if (clear)
            {
                foreach (var entity in arc.Entities)
                {
                    entity.Flags &= ~0x80u;
                }
            }

            var px = (target.PosX >> 16) + 4;
            var py = (target.PosY >> 16) + 4;
            var topZ = unchecked(target.PosZ + target.Depth);
            arc.PlaceHero(px, py, (target.PosZ >> 16) + 100);
            for (var i = 0; i < 60; i++)
            {
                arc.OneFrame();
            }

            Say($"clear={clear}: hero after the fall at ({arc.Hero.PosX >> 16},{arc.Hero.PosY >> 16}) posZ {arc.Hero.PosZ} (units {arc.Hero.PosZ >> 20}) onGround {arc.Hero.IsOnGround}; target posZ {target.PosZ} depth {target.Depth} top {topZ}; riding={(arc.Hero.RidingEntity != null)}");
        }
    }

    [Fact]
    public void PI_WhatABlockedEntityStillDoes()
    {
        using var arc = Arc14("pi");
        var tick = 0;
        var events = new List<string>();
        EntityScriptProxy? q = null;
        var done = false;
        var blocker = new CasaEngine.Framework.Scene.Entities.Entity { Name = "blocker" };
        var arrows = new List<EntityScriptProxy>();
        arc.Proxy.EventProgramRunner = new Spy(arc.Proxy.EventProgramRunner, entity =>
        {
            var idx = arrows.IndexOf(entity);
            if (idx >= 0)
            {
                events.Add($"tick {tick}: arrow {idx} (slot {entity.Slot}) program trigger {entity.EventTrigger} posX {entity.PosX >> 16}");
            }

            if (!done && ReferenceEquals(entity, q) && entity.EventTrigger == ScriptHelper.ProgramCTick)
            {
                done = true;
                events.Add($"tick {tick}: Q blocks the active others (hero included)");
                foreach (var other in arrows.Append(arc.Hero))
                {
                    if (!ReferenceEquals(other, q))
                    {
                        other.BlockedByEntity = blocker;
                    }
                }
            }
        });
        arc.Proxy.TickTraceForTests = pass =>
        {
            if (pass == "pad")
            {
                tick++;
            }

            if (done && pass == "activeEffects")
            {
                events.Add($"tick {tick} end: " + string.Join("; ", arrows.Select((a, i) => $"arrow{i} x {a.PosX >> 16} status {a.Status} anim {a.TargetAnimationId} clock {a.BoxTicks}")) + $"; hero z {arc.Hero.PosZ} x {arc.Hero.PosX >> 16}");
            }
        };

        for (var i = 0; i < 3; i++)
        {
            arrows.Add(Context(arc).SpawnEntityBySpriteIndex(arc.Hero, false, Arrow, (100 + 10 * i) << 16, 100 << 16, 0, 0x10)!);
        }

        q = arrows[1];
        arc.OneFrame();
        arc.OneFrame();
        arc.PlaceHero(150, 100, 200);
        for (var i = 0; i < 6; i++)
        {
            arc.OneFrame();
        }

        foreach (var line in events)
        {
            Say(line);
        }

        Say($"slots {string.Join(",", arrows.Select(a => a.Slot))}; hero slot {arc.Hero.Slot}; blocked hero z after: {arc.Hero.PosZ}");
    }

    [Fact]
    public void PK_ThePresetsOfTheRecipe()
    {
        var rules = Alundra.Tests.Support.SaveGameDirectorTestSupport.RealRules();
        foreach (var (name, map, tx, ty, tz) in new[] { ("pickup-m11", 11, 27, 18, 10), ("heart-m1", 1, 45, 37, 4), ("loot-m14", 14, 8, 6, 0) })
        {
            var preset = new Alundra.SaveGames.TestSaves.Preset(name, "test-" + name, map, tx, ty, tz, [], [], [], [], Hp: 4, HpMax: 10, Mp: 2, MpMax: 4);
            var ok = Alundra.SaveGames.TestSaves.TryBuild(preset, rules, out var save, out var error);
            Say($"preset {name} map {map} tile ({tx},{ty},{tz}): built {ok} {error}");
        }

        using (var arc = ArcDebug("pk11"))
        {
            arc.OneFrame();
            foreach (var e in arc.Entities.Where(x => x.SpriteType is >= 31 and <= 118))
            {
                Say($"map 11 object record {e.EntityRefId} type {e.SpriteType} px ({e.PosX >> 16},{e.PosY >> 16}) tile ({(e.PosX >> 16) / 24},{(e.PosY >> 16) / 16}) z {e.PosZ >> 20}");
            }
        }

        using (var arc = Arc14("pk14"))
        {
            arc.OneFrame();
            foreach (var e in arc.Entities.Where(x => x.SpriteType == Alundra.Ai.Enemies.BeannoideAi.SpriteType))
            {
                Say($"map 14 beannoide record {e.EntityRefId} px ({e.PosX >> 16},{e.PosY >> 16}) tile ({(e.PosX >> 16) / 24},{(e.PosY >> 16) / 16}) z {e.PosZ >> 20} contentsItem {e.ContentsItemId} flag {e.ContentsGameFlag} header {e.HeaderContents} hp {e.Hp}");
            }

            Say($"map 14 hero at px ({arc.Hero.PosX >> 16},{arc.Hero.PosY >> 16}) z {arc.Hero.PosZ >> 20}");
        }

        using (var arc = new ArcRun(new ArcSpec("E20e0c-pk1", "Overworld", "Overworld 0,0-1", Array.Empty<int>(), 45, 37, 4, 2000, RealController: true, Prefabs: true)))
        {
            arc.OneFrame();
            var heart = arc.EntityByRecord(24);
            Say($"map 1 record 24: {(heart == null ? "none" : $"type {heart.SpriteType} px ({heart.PosX >> 16},{heart.PosY >> 16}) z {heart.PosZ >> 20} status {heart.Status}")}; hero px ({arc.Hero.PosX >> 16},{arc.Hero.PosY >> 16}) z {arc.Hero.PosZ >> 20}");
        }
    }
