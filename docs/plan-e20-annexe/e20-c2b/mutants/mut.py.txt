def run(cmd):
    return subprocess.run(cmd, cwd=REPO, capture_output=True, text=True, encoding='utf-8', errors='replace', shell=False)

def apply(edits):
    for f, old, new in edits:
        p = os.path.join(REPO, f)
        b = open(p, 'rb').read().decode('utf-8')
        n = b.count(old)
        if n != 1:
            raise SystemExit('anchor count %d for %r in %s' % (n, old[:60], f))
        open(p, 'wb').write(b.replace(old, new, 1).encode('utf-8'))

def revert(edits):
    for f in sorted({e[0] for e in edits}):
        run(['git', 'checkout', '--', f])

def main(names):
    for name in names:
        edits = MUTANTS[name]
        apply(edits)
        r = run(['dotnet', 'test', 'Alundra.Tests', '-c', 'Release', '-p:CustomAfterMicrosoftCommonTargets=' + SK, '--filter', FILTER, '--blame-hang-timeout', '60s'])
        full = r.stdout + r.stderr
        revert(edits)
        clean = run(['git', 'diff', '--quiet', '--', 'Alundra', 'Alundra.Tests']).returncode == 0
        red = sorted(set(re.sub(r' \[[0-9 a-z]*\]$', '', l.strip()) for l in full.splitlines() if re.match(r'^\s*(Échoué|Failed) ', l)))
        buildfail = 'error CS' in full
        tail = [l.strip() for l in full.splitlines() if re.search(r'(Total|Réussi|Passed|Échec|Failed|Nombre total)', l) and ':' in l][-4:]
        open(os.path.join(OUT, name + '.red'), 'w', encoding='utf-8').write('\n'.join(red))
        if os.environ.get('KEEPFULL'):
            open(os.path.join(OUT, name + '.full'), 'w', encoding='utf-8').write(full)
        print('%s: rc=%d buildfail=%s killed=%s nred=%d clean=%s' % (name, r.returncode, buildfail, len(red) > 0, len(red), clean))
        for l in red[:5]:
            print('   ', l[:200])
        print('   ', tail)
        sys.stdout.flush()

if __name__ == '__main__':
    main(sys.argv[1:])
