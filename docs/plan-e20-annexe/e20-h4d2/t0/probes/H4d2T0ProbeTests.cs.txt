    private static uint Read274(EntityScriptProxy e) => PickupAi.ReadWord274(e);

    /// <summary>The numeric turn of the binary (D-H4D2-1): one step of 4, d when the numeric gap is below 4, never a wrap.</summary>
    internal static uint Turn(uint cur, uint d)
    {
        if (d == cur)
        {
            return cur;
        }

        if (d < cur)
        {
            return cur - d < 4 ? d : cur - 4;
        }

        return d - cur < 4 ? d : cur + 4;
    }

    private sealed class Run : IDisposable
    {
        internal readonly HitArena Arena;
        internal readonly Sounds Heard = new();
        internal int Tick;
        internal EntityScriptProxy? ColumnEntity;
        internal readonly List<EntityScriptProxy> Born = new();
        internal readonly Dictionary<EntityScriptProxy, EntityScriptProxy> Aim = new();
        internal readonly List<(int Tick, int Effect, int Animation, int X, int Y, int Z)> Effects = new();
        internal readonly List<string> Queries = new();
        internal readonly List<string> Notes = new();
        internal readonly StringBuilder Rows = new();
        internal readonly bool Binary;
        internal uint StateBefore;
        internal readonly List<IDisposable> Registrations = new();
        internal readonly Dictionary<int, List<(int Caller, int Count, int Results, int First)>> QueryByTick = new();

        internal Run(string name, IReadOnlyList<int[]>? blocks, Func<int, int, CutBench.Cell>? cell, int heroZ, bool binary)
        {
            Binary = binary;
            Arena = new HitArena("h4d2t0-" + name, 100, 100, new List<int[]>(), 24, 0);
            try
            {
                SyntheticField.Install(Arena.Arc, cell);
                if (heroZ != 0)
                {
                    Arena.Arc.PlaceHero(100, 100, heroZ);
                }

                var keep = new HashSet<EntityScriptProxy>(Arena.Targets) { Arena.Hero };
                foreach (var other in Arena.Arc.Entities.Where(e => !keep.Contains(e)).ToList())
                {
                    other.Status = EntityStatus.FlagToDestroy;
                }

                Arena.Arc.Proxy.RecycleDestroyedEntities();
                typeof(WorldProxy).GetField("<SoundPlayer>k__BackingField", BindingFlags.Instance | BindingFlags.NonPublic)!.SetValue(Arena.Arc.Proxy, Heard);
                Arena.Arc.Give(Item);
                Arena.Arc.SetMp(4, 4);
                PlayerManager.SetCurrentItemId(GameState.Instance, Item);
                Arena.Hero.FrameCollisionTickCounter = 0;
                Arena.Hero.DamagedTickCounter = 0;
                Arena.Hero.HitCounter = 0;
                Arena.Hero.TouchingEntity = null;
                foreach (var pixel in blocks ?? new List<int[]>())
                {
                    var block = ((IEntityWorldContext)Arena.Arc.Proxy).SpawnEntityBySpriteIndex(Arena.Hero, false, 25, pixel[0] << 16, pixel[1] << 16, 0, 0);
                    Assert.NotNull(block);
                    block!.Flags |= EntityFlags.ClassA;
                    block.Hp = 5000;
                    Arena.Targets.Add(block);
                }

                EventProgramRunner.NativeSpriteTypeFilterForTests = new HashSet<int> { Ball };
                PlayerManager.ItemStartForTests = PlayerManager.TomeStart;
                PlayerManager.ItemDriverForTests = (hero, _) => Driver(hero);
                PlayerManager.ItemCancelForTests = (hero, item) =>
                {
                    if (item == Item)
                    {
                        Cancel(hero);
                    }
                };
                Arena.Arc.Runner.NativeHandlersEnabled = false;
                var registry = Arena.Arc.Runner.NativeSprites;
                Registrations.Add(registry.Register(ScriptHelper.ProgramCTick, 63, new[] { Ball }, (entity, context, state) => C63(entity)));
                // E3 is registered for {16} in production (h4c1); the slice widens it to {16, 23} (D-H4D2-2): the probe drops the key and registers it again for both types.
                var entries = (System.Collections.IDictionary)typeof(NativeSpriteRegistry).GetField("_entries", BindingFlags.Instance | BindingFlags.NonPublic)!.GetValue(registry)!;
                entries.Remove((ScriptHelper.ProgramEDeactivate, 3));
                Registrations.Add(registry.Register(ScriptHelper.ProgramEDeactivate, 3, new[] { 16, Ball }, (entity, context, state) => EarthTome1Ai.ImpactDeactivate(entity)));
                NativeEffects.CreatedForTests = slot => Effects.Add((Tick, slot.TargetEffect, slot.TargetAnimation, slot.X, slot.Y, slot.Z));
            }
            catch
            {
                Dispose();
                throw;
            }
        }

        private IEntityWorldContext World => (IEntityWorldContext)Arena.Arc.Proxy;

        private int CountOfTheTable()
        {
            var highest = 0;
            foreach (var e in World.SpawnedEntities)
            {
                if (e.Slot >= 1 && e.Status != EntityStatus.Destroyed && e.Status != EntityStatus.FlagToDestroy && e.Slot > highest)
                {
                    highest = e.Slot;
                }
            }

            return highest;
        }

        // the driver 0x80035A84 (throwaway port)
        private int Driver(EntityScriptProxy hero)
        {
            var phase = hero.ItemUsePhase;
            if (phase == 0)
            {
                var guard = PlayerManager.TomeSlopeGuard(hero);
                if (guard != 0)
                {
                    return guard;
                }

                if (hero.TargetAnimationId != 0x33)
                {
                    return 0;
                }

                ColumnEntity = World.SpawnEntityBySpriteIndex(hero, false, Column, hero.PosX, hero.PosY, hero.PosZ, 0);
                if (ColumnEntity is null)
                {
                    Notes.Add("column refused at tick " + Tick);
                    return 1;
                }

                Born.Add(ColumnEntity);
                hero.ItemUsePhase = 1;
                Notes.Add("column born tick " + Tick + " counter " + hero.ItemUseCounter + " slot " + ColumnEntity.Slot + " status " + (int)ColumnEntity.Status);
                phase = 1;
            }

            if (phase == 1)
            {
                var guard = PlayerManager.TomeSlopeGuard(hero);
                if (guard != 0)
                {
                    return 1;
                }

                var c = ColumnEntity!;
                if (c.Status == EntityStatus.Destroyed || c.Slot < 0)
                {
                    hero.TargetAnimationId = 0x34;
                    hero.ItemUsePhase = 2;
                    Notes.Add("identity changed at tick " + Tick + " counter " + hero.ItemUseCounter);
                    return 0;
                }

                if ((hero.ItemUseCounter & 7) != 0)
                {
                    return 0;
                }

                var direction = EnemyHelpers.Below(32);
                var ball = World.SpawnEntityBySpriteIndex(hero, false, Ball, hero.PosX, hero.PosY, hero.PosZ + 0x100000, direction);
                if (ball is not null)
                {
                    ball.ForceZ = 0x80000;
                    Write274(ball, 0x10);
                    ball.DelayOrAngleOrEntityId = 0;
                    Born.Add(ball);
                    Notes.Add("ball born tick " + Tick + " slot " + ball.Slot + " dir " + direction);
                }
                else
                {
                    Notes.Add("ball refused tick " + Tick + " dir " + direction);
                }

                return 0;
            }

            if (phase == 2)
            {
                return hero.TargetAnimationId != 0x34 ? 1 : 0;
            }

            return 0;
        }

        // the cancellation 0x80034D2C (throwaway port), reached by the guard only
        private void Cancel(EntityScriptProxy hero)
        {
            var count = CountOfTheTable();
            foreach (var e in World.SpawnedEntities.ToList())
            {
                if (e.Slot < 1 || e.Slot > count || e.Status != EntityStatus.Normal)
                {
                    continue;
                }

                if (e.SpriteType == Column)
                {
                    var x = e.PosX;
                    var y = e.PosY;
                    var z = e.PosZ + 1;
                    Arena.Arc.Proxy.EffectStage!.Emit(reservoir => NativeEffects.CreateFireColumnBlast(reservoir, x, y, z));
                    World.DestroyEntity(e);
                    Notes.Add("cancel destroys the column at tick " + Tick);
                }
                else if (e.SpriteType == Ball)
                {
                    e.Status = EntityStatus.Deactivated;
                    e.BlockedByEntity = null;
                    Notes.Add("cancel deactivates a ball (slot " + e.Slot + ") at tick " + Tick);
                }
            }
        }

        // C63 0x8007AC60 (throwaway port)
        private void C63(EntityScriptProxy e)
        {
            var wait = Read274(e);
            if (wait != 0)
            {
                Write274(e, wait - 1);
                return;
            }

            e.ForceZ = unchecked((int)0xFFFD0000);
            if (e.DelayOrAngleOrEntityId == 0)
            {
                var count = CountOfTheTable();
                var type = (e.CurrentAttack?.Attribute ?? 0) & 0xF;
                var found = TargetQuery.Find(World.SpawnedEntities, count, e, 1, type, !Binary ? true : false, !Binary ? 2 : 3);
                Queries.Add(string.Join(",", Tick, e.Slot, count, found.Count, found.Count > 0 ? found[0].Slot : 0, type));
                if (found.Count == 0)
                {
                    Write274(e, (uint)EnemyHelpers.Below(9));
                    return;
                }

                Aim[e] = found[0];
                e.DelayOrAngleOrEntityId = found[0].Slot;
                DoTurn(e, found[0]);
                return;
            }

            if (!Aim.TryGetValue(e, out var target) || target.Status != EntityStatus.Normal || target.Slot != e.DelayOrAngleOrEntityId)
            {
                e.DelayOrAngleOrEntityId = 0;
                return;
            }

            DoTurn(e, target);
        }

        private static void DoTurn(EntityScriptProxy e, EntityScriptProxy target)
        {
            var d = ScriptHelper.GetDirectionToTarget(unchecked(target.PosX - e.PosX), unchecked(target.PosY - e.PosY));
            var cur = e.TargetDirection;
            if (d == cur)
            {
                return;
            }

            e.TargetDirection = Turn(cur, d);
            Write274(e, 2);
        }

        internal static uint State() => unchecked((uint)GameRandom.RandomSeed);

        internal static int Steps(uint from, uint to)
        {
            var s = from;
            for (var k = 0; k <= 64; k++)
            {
                if (s == to)
                {
                    return k;
                }

                s = unchecked(s * 0x7D2B89DDu + 0xE06A02E7u);
            }

            return -1;
        }

        private static void Num(StringBuilder b, long v) => b.Append(v.ToString(CultureInfo.InvariantCulture));

        internal void Go(int last, int? hurt = null, int ticksPerFrame = 1, Dictionary<int, int[]>? remove = null, uint seed = 0x12345678)
        {
            Arena.BeforeTick = tick =>
            {
                Tick = tick;
                Heard.Tick = tick;
                if (tick == 1)
                {
                    GameRandom.RandomSeed = seed;
                }

                if ((tick - 1) % ticksPerFrame == 0)
                {
                    StateBefore = State();
                }

                if (hurt == tick)
                {
                    Arena.Hero.TargetAnimationId = 0x31;
                }

                if (remove != null && remove.TryGetValue(tick, out var gone))
                {
                    foreach (var index in gone)
                    {
                        Arena.Targets[index].Status = EntityStatus.FlagToDestroy;
                    }
                }
            };
            Arena.AfterTick = tick =>
            {
                var b = new StringBuilder();
                b.Append("{\"t\":").Append(tick).Append(",\"hero\":[");
                var h = Arena.Hero;
                b.Append(string.Join(",", h.TargetAnimationId, h.PosX, h.PosY, h.PosZ, h.ItemInUse, h.ItemUsePhase, h.ItemUseCounter, GameState.Instance.PlayerStats.Mp));
                b.Append("],\"ents\":[");
                var first = true;
                foreach (var e in Born)
                {
                    var live = e.Status != EntityStatus.Destroyed;
                    if (live)
                    {
                        e.EvaluateFrameBox();
                    }

                    if (!first)
                    {
                        b.Append(',');
                    }

                    first = false;
                    b.Append('[').Append(string.Join(",", e.SpriteType, e.Slot, (int)e.Status, e.PosX, e.PosY, e.PosZ, e.ForceX, e.ForceY, e.ForceZ, (long)e.Flags, e.TargetAnimationId, e.TargetDirection, e.ForceResetAnimationFlag,
                        Read274(e), e.DelayOrAngleOrEntityId, e.IsOnGround, e.Slope_18c, e.CurrentAttack?.Attribute ?? -1, e.CurrentAttack?.Power ?? -1, e.FrameCollisionTickCounter,
                        live ? e.CollisionWidth : -1, live ? e.CollisionDepth : -1, live ? e.CollisionHeight : -1, live ? e.CollisionOffsetX : -1, live ? e.CollisionOffsetY : -1, live ? e.CollisionOffsetZ : -1)).Append(']');
                }

                b.Append("],\"targets\":[");
                b.Append(string.Join(",", Arena.Targets.Select(t => "[" + t.FrameCollisionTickCounter + "," + (t.TouchingEntity?.GameplayProxy is EntityScriptProxy p ? p.SpriteType : -1) + "," + (int)t.Status + "]")));
                b.Append("],\"state\":").Append(State()).Append('}');
                Rows.AppendLine(b.ToString());
            };

            Arena.Begin();
            var frames = new StringBuilder();
            for (var tick = 1; tick <= last; tick += ticksPerFrame)
            {
                var ticks = Math.Min(ticksPerFrame, last - tick + 1);
                Arena.Frame(tick == 1 ? PadState.Circle : 0, ticks);
                frames.AppendLine(string.Join(",", tick + ticks - 1, State(), Steps(StateBefore, State())));
            }

            Gen = frames.ToString();
        }

        internal string Gen = "";

        internal void Write(string name)
        {
            File.WriteAllText(Path.Combine(Out, name + ".rows.jsonl"), Rows.ToString());
            File.WriteAllText(Path.Combine(Out, name + ".gen.csv"), Gen);
            File.WriteAllText(Path.Combine(Out, name + ".queries.csv"), string.Join("\n", Queries) + "\n");
            File.WriteAllText(Path.Combine(Out, name + ".effects.csv"), string.Join("\n", Effects.Select(e => string.Join(",", e.Tick, e.Effect, e.Animation, e.X, e.Y, e.Z))) + "\n");
            File.WriteAllText(Path.Combine(Out, name + ".sounds.csv"), string.Join("\n", Heard.Played.Select(p => p.Tick + "," + p.Sfx)) + "\n");
            File.WriteAllText(Path.Combine(Out, name + ".notes.txt"), string.Join("\n", Notes) + "\n");
        }

        public void Dispose()
        {
            foreach (var r in Registrations)
            {
                r.Dispose();
            }

            PlayerManager.ItemStartForTests = null;
            PlayerManager.ItemDriverForTests = null;
            PlayerManager.ItemCancelForTests = null;
            EventProgramRunner.NativeSpriteTypeFilterForTests = null;
            NativeEffects.CreatedForTests = null;
            Arena.Dispose();
        }
    }

    [Fact]
    public void P15TurnGrid()
    {
        if (VecC63.Length == 0)
        {
            return;
        }

        using var doc = JsonDocument.Parse(File.ReadAllText(VecC63));
        var lines = new List<string>();
        var bad = 0;
        var badTurn = 0;
        var n = 0;
        foreach (var v in doc.RootElement.GetProperty("vectors").EnumerateArray())
        {
            if (v.GetProperty("kind").GetString() != "turn")
            {
                continue;
            }

            var reads = new Dictionary<long, long>();
            foreach (var r in v.GetProperty("reads").EnumerateArray())
            {
                reads[r[0].GetInt64()] = r[2].GetInt64();
            }

            var self = v.GetProperty("args")[0].GetInt64();
            int Word(long a) => unchecked((int)(uint)reads[a]);
            var ptr = reads[self + 0x278];
            var sx = Word(self + 0x114);
            var sy = Word(self + 0x118);
            var tx = Word(ptr + 0x114);
            var ty = Word(ptr + 0x118);
            var d = ScriptHelper.GetDirectionToTarget(unchecked(tx - sx), unchecked(ty - sy));
            var expected = (uint)v.GetProperty("target_dir").GetInt64();
            var cur = (uint)v.GetProperty("cur").GetInt64();
            var written = cur;
            foreach (var w in v.GetProperty("writes").EnumerateArray())
            {
                if (w[0].GetInt64() == self + 0x8C)
                {
                    written = (uint)w[2].GetInt64();
                }
            }

            n++;
            if (d != expected)
            {
                bad++;
                lines.Add("dir mismatch id " + v.GetProperty("id").GetInt32() + " cur " + cur + " computed " + d + " binary " + expected);
            }

            if (Turn(cur, expected) != written)
            {
                badTurn++;
                lines.Add("turn mismatch id " + v.GetProperty("id").GetInt32() + " cur " + cur + " d " + expected + " model " + Turn(cur, expected) + " binary " + written);
            }
        }

        lines.Insert(0, "turn-grid vectors " + n + " direction mismatches " + bad + " turn mismatches " + badTurn);
        File.WriteAllText(Path.Combine(Out, "p15.txt"), string.Join("\n", lines) + "\n");
    }
