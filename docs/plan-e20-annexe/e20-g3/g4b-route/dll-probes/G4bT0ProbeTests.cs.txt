    private static void Out(string line) => File.AppendAllText(OutPath, line + "\n", new UTF8Encoding(false));

    private static ArcSpec Spec(string name) => new(
        name, "The Klark", "Ship Klark (night, inner, break)-392", new[] { 1641 }, 29, 13, 4, 1_000_000, RealController: true, Prefabs: true);

    private sealed class Rig
    {
        public readonly ArcRun Arc;
        public readonly int K;
        public int Anchor = -1;
        public int Tick = -1;
        public readonly List<int> ActualPolls = new();
        public int Rolls;
        public readonly List<int> RollTicks = new();
        public Action<int>? AfterTick;
        private readonly float _elapsed;

        public Rig(int k, string name)
        {
            K = k;
            _elapsed = k / 50f;
            Arc = new ArcRun(Spec(name));
            Arc.Proxy.TickTraceForTests = pass =>
            {
                if (pass == "pad") { Tick++; }
                else if (pass == "inventory") { AfterTick?.Invoke(Tick); }
            };
            Arc.OnInstruction = t =>
            {
                if (t.Slot != B) { return; }
                if (t.Pc == 104 && t.Opcode == 0x11 && Anchor < 0) { Anchor = t.Frame; }
                if (t.Pc == 107 && t.Opcode == 0x2F) { ActualPolls.Add(t.Frame); }
                if (t.Pc == 118 && t.Opcode == 0x10) { Rolls++; RollTicks.Add(Tick); }
            };
            while (Anchor < 0) { Arc.OneFrame(_elapsed); }
            for (var n = 0; n < 12; n++) { _recs[n] = Arc.EntityByRecord(n)!; }
        }

        private readonly EntityScriptProxy[] _recs = new EntityScriptProxy[12];

        public bool IsPoll(int image)
        {
            if (image <= Anchor) { return false; }
            for (var m = 1; ; m++)
            {
                var p = Anchor + 16 * m / K;
                if (p == image) { return true; }
                if (p > image) { return false; }
            }
        }

        public uint LastDirs; public bool LastImagePoll;

        public bool InPoll => IsPoll(Arc.Frame);

        public bool PreviousWasPoll => IsPoll(Arc.Frame - 1);

        /// <summary>One image of the pilot: in a poll image the directions and the buttons are let go (returns false then).</summary>
        public bool Step(uint dirs, uint buttons = 0, bool obey = true)
        {
            var poll = obey && IsPoll(Arc.Frame);
            if (poll) { dirs = 0; buttons = 0; }
            LastDirs = dirs; LastImagePoll = poll;
            Arc.HoldDirections(dirs);
            if (buttons != 0) { Arc.Press(buttons, _elapsed); } else { Arc.OneFrame(_elapsed); }
            return !poll;
        }

        public void ToAfterPoll()
        {
            do { Step(0); } while (!PreviousWasPoll);
        }

        public double X => Arc.Hero.PosX / 65536.0;
        public double Y => Arc.Hero.PosY / 65536.0;
        public double Z => Arc.Hero.PosZ / 65536.0;
        public bool OnGround => Arc.Hero.IsOnGround != 0;

        public void Place(double x, double y, int z)
        {
            Arc.PlaceHero((int)Math.Floor(x), (int)Math.Floor(y), z);
            Arc.Hero.PosX = (int)Math.Round(x * 65536.0);
            Arc.Hero.PosY = (int)Math.Round(y * 65536.0);
            Arc.Hero.PushLogicalPositionToRoot();
            var root = new Microsoft.Xna.Framework.Vector3((float)x, (float)y, z);
            Arc.HeroEntity.RootComponent!.LocalTransform.Position = root;
            Arc.Hero.Controller!.Teleport(root);
        }

        public EntityScriptProxy Rec(int record) => _recs[record];

        public bool Touches(int record) => ReferenceEquals(Arc.Hero.XCollisionEntity, Rec(record));

        public int? Carried()
        {
            var c = Arc.Hero.CarriedEntity;
            if (c == null) { return null; }
            for (var r = 0; r < 12; r++)
            {
                if (ReferenceEquals(c, _recs[r].LogicContextEntity)) { return r; }
            }
            return -1;
        }

        public string PollCheck()
        {
            var predicted = Enumerable.Range(Anchor + 1, Math.Max(0, Arc.Frame - Anchor)).Where(IsPoll).ToList();
            var ok = predicted.Take(ActualPolls.Count).SequenceEqual(ActualPolls) && ActualPolls.All(IsPoll);
            return ok ? $"polls predicted = actual ({ActualPolls.Count})" : $"POLL MISMATCH actual [{string.Join(",", ActualPolls)}] vs predicted [{string.Join(",", predicted)}]";
        }

        public void Close()
        {
            try { Arc.Dispose(); } catch (Xunit.Sdk.XunitException) { /* the slide guard of Dispose: a probe measures the count */ }
        }
    }

    private sealed class Log
    {
        public readonly List<string> Lines = new();
        public void Add(Rig r, string s) => Lines.Add($"  image {r.Arc.Frame} tick {r.Tick + 1}: {s} at {Fmt(r)}");
    }

    private static void SlideWatch(Rig r, List<string> events, Func<string> phase)
    {
        var last = 0;
        var prevContact = "none";
        r.AfterTick = tick =>
        {
            var sl = r.Arc.TotalSlideCount;
            if (sl != last) { events.Add($"tick {tick} slides {last}->{sl} at {Fmt(r)} phase {phase()} contact at the end of the previous tick: {prevContact}"); last = sl; }
            var c = r.Arc.Hero.XCollisionEntity;
            prevContact = c == null ? "none" : "rec" + Enumerable.Range(0, 12).FirstOrDefault(n => ReferenceEquals(c, r.Rec(n)), -1);
        };
    }

    /// <summary>(b1) the lift route: down, left, lift rec5, carry, throw, then as the jump route.</summary>
    [Theory]
    [InlineData(1, 615.0)]
    [InlineData(2, 615.0)]
    [InlineData(3, 615.0)]
    public void B1_Route(int k, double throwX)
    {
        var r = new Rig(k, "g4b-b1-" + k);
        try
        {
            var phase = "down";
            var log = new Log();
            var slideEvents = new List<string>();
            SlideWatch(r, slideEvents, () => phase);
            var liftTick = -1; var thrownTick = -1; var destroyedTick = -1; var carryStandTick = -1; var liftImage = -1; var throwImage = -1;
            var prevAfter = r.AfterTick;
            r.AfterTick = tick =>
            {
                prevAfter?.Invoke(tick);
                var c = r.Carried();
                if (liftTick < 0 && c == 5) { liftTick = tick; }
                if (carryStandTick < 0 && c == 5 && r.Arc.Hero.TargetAnimationId == 12) { carryStandTick = tick; }
                if (thrownTick < 0 && throwImage >= 0 && c == null) { thrownTick = tick; }
                if (destroyedTick < 0 && throwImage >= 0 && r.Rec(5).Status == EntityStatus.Destroyed) { destroyedTick = tick; }
            };
            var images = 0;
            while (!WarpDirector.Instance.HasPendingArrival && images < 1500 && (r.Rolls == 0))
            {
                images++;
                var x = r.X; var y = r.Y;
                void Go(string to) { log.Add(r, $"{phase} -> {to}"); phase = to; }
                if (phase == "down" && y >= 288.0) { Go("left"); }
                if (phase == "south" && y >= 343.5) { Go("west"); }
                if (phase == "west" && x <= 541.0) { Go("down2"); }
                if (phase == "after_throw" && x <= 600.0) { Go("south"); }
                var hold = 0u; uint btn = 0;
                switch (phase)
                {
                    case "down" or "south" or "down2": hold = PadState.Down; break;
                    case "west" or "after_throw": hold = PadState.Left; break;
                    case "left":
                        hold = PadState.Left;
                        if (r.Touches(5) && !r.InPoll) { btn = PadState.Square; liftImage = r.Arc.Frame; log.Add(r, "Square (lift) pressed, contact rec5"); Go("gesture"); }
                        break;
                    case "gesture":
                        hold = PadState.Left;
                        if (r.Carried() == 5 && r.Arc.Hero.TargetAnimationId == 12) { Go("carry"); goto case "carry"; }
                        break;
                    case "carry":
                        hold = PadState.Left;
                        if (x <= throwX && !r.InPoll) { btn = PadState.Square; throwImage = r.Arc.Frame; log.Add(r, "Square (throw) pressed"); Go("after_throw"); }
                        break;
                }

                r.Step(hold, btn);
            }

            Out($"== B1 lift route k={k} throwX={throwX}: images {images}, frame {r.Arc.Frame}, tick {r.Tick}, arrival={WarpDirector.Instance.HasPendingArrival}, rolls {r.Rolls}, slides {r.Arc.TotalSlideCount}, final {Fmt(r)}, phase {phase}");
            Out($"   lift: Square image {liftImage}, carried at the end of tick {liftTick}, carry stand at tick {carryStandTick} ; throw image {throwImage}, hands free at tick {thrownTick}, rec5 destroyed at tick {destroyedTick}");
            foreach (var l in log.Lines) { Out(l); }
            foreach (var l in slideEvents) { Out("  SLIDE " + l); }
            Out($"   {r.PollCheck()} ; departure map {(WarpDirector.Instance.HasPendingArrival ? WarpDirector.Instance.ArrivalRecordForTests.MapIndex : 0u)}");
        }
        finally { r.Close(); }
    }

            {
                if (r.Touches(5) && contactSeenAt < 0) { contactSeenAt = r.Arc.Frame; }
                if (r.Touches(5) && !r.InPoll) { sqImage = r.Arc.Frame; yAtSquare = r.Y; r.Step(PadState.Up, PadState.Square); }
                else { r.Step(PadState.Up); }
            }
            var carriedAfter = r.Carried();
            var guardB5 = 0;
            while (r.Arc.Hero.TargetAnimationId != 12 && guardB5++ < 100) { r.Step(0); }
            foreach (var l in tickLines.Where(l => int.Parse(l.Split(':')[0].Replace("   tick ", "")) >= 205 && int.Parse(l.Split(':')[0].Replace("   tick ", "")) <= 232)) { Out(l); }
            Out($"== B5 k={k}: contact first seen at the start of image {contactSeenAt}, Square image {sqImage} (y {yAtSquare:F4}), carried right after {carriedAfter}, hero {Fmt(r)}, carried at the end {r.Carried()}, anim {r.Arc.Hero.TargetAnimationId}, slides {r.Arc.TotalSlideCount}, rolls {r.Rolls}");
            foreach (var l in slideEvents) { Out("  SLIDE " + l); }
        }
        finally { r.Close(); }
    }

    /// <summary>(b6) the roll while carrying: lift rec5 as (b5), then at the first poll image after the carry stand hold Left through the poll (the pilot breaks its own rule 2 on purpose) and watch the hands for 12 ticks after the roll.</summary>
    [Theory]
    [InlineData(1)]
    [InlineData(2)]
    [InlineData(3)]
    public void B6_Roll(int k)
    {
        var r = new Rig(k, "g4b-b6-" + k);
        try
        {
            r.Place(657.0, 312.0, 64);
            var slideEvents = new List<string>();
            SlideWatch(r, slideEvents, () => "b6");
            var sqImage = -1;
            for (var i = 0; i < 400 && sqImage < 0; i++)
            {
                if (r.Touches(5) && !r.InPoll) { sqImage = r.Arc.Frame; r.Step(PadState.Up, PadState.Square); }
                else { r.Step(PadState.Up); }
            }

            var guard = 0;
            while (r.Arc.Hero.TargetAnimationId != 12 && guard++ < 200) { r.Step(0); }
            var standTick = r.Tick;
            Out($"== B6 k={k}: lifted at image {sqImage}, carry stand reached at tick {standTick} (frame {r.Arc.Frame}), carried {r.Carried()}");
            var rows = new List<string>();
            var prevAfter = r.AfterTick;
            var rollTick = -1;
            r.AfterTick = tick =>
            {
                prevAfter?.Invoke(tick);
                if (rollTick < 0 && r.Rolls > 0) { rollTick = tick; }
                if (rollTick >= 0 && tick <= rollTick + 12) { rows.Add($"tick {tick} (roll+{tick - rollTick}): hero {Fmt(r)} anim {r.Arc.Hero.TargetAnimationId} carried {r.Carried()} rec5.z {r.Rec(5).PosZ / 65536.0:F2}"); }
            };
            guard = 0;
            while (!r.InPoll && guard++ < 200) { r.Step(0); }
            Out($"   the poll image is {r.Arc.Frame} (tick {r.Tick + 1}); Left held in it (the rule 2 broken on purpose) and in the following images");
            r.Step(PadState.Left, 0, obey: false);
            guard = 0;
            while ((rollTick < 0 || r.Tick <= rollTick + 12) && guard++ < 100) { r.Step(PadState.Left); }
            r.AfterTick = prevAfter;
            foreach (var l in rows) { Out("   " + l); }
            Out($"   end: hero {Fmt(r)} anim {r.Arc.Hero.TargetAnimationId} carried {r.Carried()} rolls {r.Rolls} at ticks [{string.Join(",", r.RollTicks)}], slides {r.Arc.TotalSlideCount}, rec5 status {r.Rec(5).Status}");
            foreach (var l in slideEvents) { Out("  SLIDE " + l); }
        }
        finally { r.Close(); }
    }
