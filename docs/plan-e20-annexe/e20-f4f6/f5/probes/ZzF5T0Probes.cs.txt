    [Fact]
    public void P3_Generator_AfterEachFrame_EqualsTheKeyWithTheStepByImage()
    {
        foreach (var (name, item, cell, targets) in new (string, int, Func<int, int, CutBench.Cell>?, IReadOnlyList<int[]>?)[]
        {
            ("arrow3_east_wall", 5, Wall15, null),
            ("arrow3_target", 5, null, new List<int[]> { new[] { 300, 104 } }),
        })
        {
            var key = ArrowOracle.Load(name, stepByImage: true);
            using var run = new ArrowRun("p3-" + name, 5, cell, targets: targets, generatorAtTick1: key.Rng0(1));
            if (targets is not null)
            {
                run.Arena.Targets[0].Depth = (32 << 16) - 1;
            }

            run.Go(1, key.Ticks);
            var bad = new List<string>();
            for (var t = 1; t < key.Ticks; t++)
            {
                if (run.StateAfter[t] != key.Rng0(t + 1) || run.StepsDuring[t] != key.Draws(t) + 1)
                {
                    bad.Add($"t{t}: state 0x{run.StateAfter[t]:X8} (key 0x{key.Rng0(t + 1):X8}), steps {run.StepsDuring[t]} (key draws {key.Draws(t)} + 1)");
                }
            }

            Log.AppendLine($"P3 {name}: generator mismatches {bad.Count}" + (bad.Count > 0 ? ": " + string.Join(" | ", bad.Take(6)) : string.Empty));
        }
    }

    [Fact]
    public void P5_TheBounce_AtOneTwoThreeTicksPerFrame_BeforeTheSpeedRule()
    {
        var key = ArrowOracle.Load("arrow3_east_wall");
        foreach (var cadence in new[] { 1, 2, 3 })
        {
            using var run = new ArrowRun("p5-" + cadence, 5, Wall15);
            run.Go(cadence, 90);
            var bounceTick = Enumerable.Range(0, 80).FirstOrDefault(k => run.Ticks.TryGetValue(run.BirthTick + k, out var t) && t.Animation == 1, -1);
            Log.AppendLine($"P5 cadence {cadence}: born {run.BirthTick}, bounce (animation 1) first seen at S+{bounceTick} (key: S+{key.FirstTickWhere("anim_t")}), last present S+{Enumerable.Range(0, 80).Where(k => run.Ticks.TryGetValue(run.BirthTick + k, out var t) && t.Present).Max()} (key S+{key.LastTickPresent()})");
            var sb = new StringBuilder("   dx(DLL-key, px) from S+40:");
            for (var k = 40; k <= 62; k++)
            {
                if (run.Ticks.TryGetValue(run.BirthTick + k, out var t) && t.Present && key.Arrow(key.FirstTick + k) is { } a)
                {
                    sb.Append($" {k}:{(t.X - a.GetProperty("px").GetInt32()) / 65536.0:F3}");
                }
            }

            Log.AppendLine(sb.ToString());
        }
    }

    private static void Describe(string label, ArrowRun run, ArrowOracle key)
    {
        var s = run.BirthTick;
        var hit = run.Hits.Where(h => h.Value.HitCounter != 0).Select(h => h.Key - s).DefaultIfEmpty(-1).Min();
        var gone = Enumerable.Range(0, 80).Where(k => !run.Ticks.TryGetValue(s + k, out var t) || !t.Present).Select(k => k).DefaultIfEmpty(-1).Min();
        var dead = Enumerable.Range(0, 80).Where(k => run.Ticks.TryGetValue(s + k, out var t) && t.Status == EntityStatus.FlagToDestroy).Select(k => k).DefaultIfEmpty(-1).Min();
        var sparks = run.Hits.Values.Sum(h => h.Sparks);
        Log.AppendLine($"P6 {label}: born {s}, first hit S+{hit} (key S+{key.FirstTickWhere("hitc")}), first status 4 S+{dead}, first absent S+{gone} (key last present S+{key.LastTickPresent()}), spark sets {sparks}, target touched at {string.Join(",", run.TargetTouchedTicks.Select(t => "S+" + (t - s)))}");
    }

    [Fact]
    public void P6_TheTargets_BeforeTheRule()
    {
        foreach (var (name, depth) in new[] { ("arrow3_target", 32), ("arrow3_zreach_16", 16), ("arrow3_zreach_17", 17), ("arrow3_target_low", 8) })
        {
            using var run = new ArrowRun("p6-" + name, 5, null, targets: new List<int[]> { new[] { 300, 104 } });
            run.Arena.Targets[0].Depth = (depth << 16) - 1;
            run.Go(1, 80);
            Describe(name + " (body depth " + depth + ")", run, ArrowOracle.Load(name));
        }

        using (var run = new ArrowRun("p6-136", 5, null, alsoKeep: e => e.SpriteType == 0x100 + 136))
        {
            var kept = run.Kept;
            Log.AppendLine($"P6 sprite 136 kept: {kept is not null}; body (w,h,d) = ({kept?.Width},{kept?.Height},{kept?.Depth}) mod=({kept?.ModX},{kept?.ModY},{kept?.ModZ}) flags=0x{kept?.Flags:X} animation {kept?.TargetAnimationId} hasBox {kept?.HasFrameBox}");
            if (kept is not null)
            {
                HitArena.Place(kept, 300, 104);
                kept.TargetAnimationId = 3;
                run.Go(1, 80);
                Describe("arrow3_target_sprite136", run, ArrowOracle.Load("arrow3_target_sprite136"));
                Log.AppendLine($"   after: animation {kept.TargetAnimationId} hasBox {kept.HasFrameBox} box=({kept.CollisionWidth},{kept.CollisionDepth},{kept.CollisionHeight}) frameCollisionCounter {kept.FrameCollisionTickCounter}");
            }
        }
    }
