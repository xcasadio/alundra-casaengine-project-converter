def run(cmd):
    return subprocess.run(cmd, cwd=REPO, capture_output=True, text=True, encoding='utf-8', errors='replace', shell=False)

def apply(edits):
    for f, old, new in edits:
        p = os.path.join(REPO, f)
        b = open(p, 'rb').read().decode('utf-8')
        if '\r\n' in b:
            old = old.replace('\r\n', '\n').replace('\n', '\r\n')
            new = new.replace('\r\n', '\n').replace('\n', '\r\n')
        n = b.count(old)
        if n != 1:
            raise SystemExit('anchor count %d for %s in %s' % (n, old[:50], f))
        open(p, 'wb').write(b.replace(old, new, 1).encode('utf-8'))

def revert(edits):
    for f in sorted({e[0] for e in edits}):
        run(['git', 'checkout', '--', f])
